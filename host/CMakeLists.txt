cmake_minimum_required(VERSION 3.20)
project(sma3_probe C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(SMA3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(GBARECOMP_GENERATED_BIOS_DIR "${SMA3_ROOT}/build/generated-bios" CACHE PATH "" FORCE)
set(GBARECOMP_COMPILER_CACHE OFF CACHE STRING "" FORCE)
add_subdirectory("${SMA3_ROOT}/sources/gbarecomp" framework EXCLUDE_FROM_ALL)
add_library(sma3_cartridge STATIC IMPORTED)
set_target_properties(sma3_cartridge PROPERTIES IMPORTED_LOCATION "${SMA3_ROOT}/build/cartridge/libgbarecomp_game.a")
add_executable(sma3_probe main.cpp)
target_include_directories(sma3_probe PRIVATE "${SMA3_ROOT}/sources/gbarecomp/src/runtime")
target_link_libraries(sma3_probe PRIVATE sma3_cartridge gbarecomp_runtime gbarecomp_debug)
